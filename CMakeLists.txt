cmake_minimum_required(VERSION 3.21)
project(zarc VERSION 0.1.0 LANGUAGES C)
set(CMAKE_C_STANDARD 17)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(zarc_domain src/domain.c)
target_include_directories(zarc_domain PUBLIC include)
function(zarc_warnings target)
  if(MSVC)
    target_compile_options(${target} PRIVATE /W4 /WX)
  else()
    target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
  endif()
endfunction()
zarc_warnings(zarc_domain)
add_library(zarc_composition src/composition.c)
target_include_directories(zarc_composition PUBLIC include)
zarc_warnings(zarc_composition)
add_library(zarc_supervision src/health.c src/diagnostics.c src/config_store.c)
target_include_directories(zarc_supervision PUBLIC include)
zarc_warnings(zarc_supervision)
add_library(zarc_connect src/connect.c)
target_include_directories(zarc_connect PUBLIC include)
zarc_warnings(zarc_connect)
add_library(zarc_controller src/controller.c)
target_include_directories(zarc_controller PUBLIC include)
target_link_libraries(zarc_controller PUBLIC zarc_domain zarc_connect)
zarc_warnings(zarc_controller)
add_library(zarc_coordination src/update.c src/power.c)
target_include_directories(zarc_coordination PUBLIC include)
zarc_warnings(zarc_coordination)
add_executable(zarc_controller_pc apps/controller_pc.c)
target_link_libraries(zarc_controller_pc PRIVATE zarc_domain)
zarc_warnings(zarc_controller_pc)
add_library(zarc_loopback platform/desktop/loopback.c)
target_include_directories(zarc_loopback PUBLIC platform/desktop)
target_link_libraries(zarc_loopback PUBLIC zarc_connect)
if(WIN32)
  target_link_libraries(zarc_loopback PRIVATE ws2_32)
endif()
zarc_warnings(zarc_loopback)
add_executable(zarc_controller_lab_pc apps/controller_lab_pc.c)
target_link_libraries(zarc_controller_lab_pc PRIVATE zarc_controller zarc_loopback)
zarc_warnings(zarc_controller_lab_pc)
add_executable(zarc_broker_pc apps/broker_lab_pc.c)
target_link_libraries(zarc_broker_pc PRIVATE zarc_loopback)
zarc_warnings(zarc_broker_pc)
include(CTest)

  add_executable(zarc_contract_tests tests/contract_tests.c)
  target_link_libraries(zarc_contract_tests PRIVATE zarc_domain)
  zarc_warnings(zarc_contract_tests)

  zarc_warnings(zarc_composition_tests)

  zarc_warnings(zarc_supervision_tests)
  add_test(NAME supervision_contract COMMAND zarc_supervision_tests)
  add_executable(zarc_connect_tests tests/connect_tests.c)
  target_link_libraries(zarc_connect_tests PRIVATE zarc_connect)
  zarc_warnings(zarc_connect_tests)
  add_test(NAME connect_contract COMMAND zarc_connect_tests)
  add_executable(zarc_controller_tests tests/controller_tests.c)
  target_link_libraries(zarc_controller_tests PRIVATE zarc_controller)
  zarc_warnings(zarc_controller_tests)
  add_test(NAME controller_contract COMMAND zarc_controller_tests)
  add_executable(zarc_coordination_tests tests/coordination_tests.c)
  target_link_libraries(zarc_coordination_tests PRIVATE zarc_coordination)
  zarc_warnings(zarc_coordination_tests)
  add_test(NAME coordination_contract COMMAND zarc_coordination_tests)
